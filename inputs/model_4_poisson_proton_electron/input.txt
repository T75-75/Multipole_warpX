############################################
### MODEL 4: POISSON ELECTRODE PROTON/ELECTRON CLOUD
############################################

# Project transverse coordinates are (x, y). WarpX 2D Cartesian uses (x, z),
# so project y is mapped to WarpX z.
# No external particle field is applied here. Electrode and plasma fields are
# solved together by the electrostatic Poisson solve.
# The timestep is limited from the electron plasma frequency so electron
# oscillations are resolved.

#################################
########### CONSTANTS ###########
#################################
my_constants.r0 = 1.0e-3
my_constants.sigma_r = 3.333333333333333e-4
my_constants.ion_temperature_eV = 100.0
my_constants.electron_temperature_eV = 5.0
my_constants.vth_over_c = sqrt(ion_temperature_eV*q_e/m_p)/clight
my_constants.electron_vth_over_c = sqrt(electron_temperature_eV*q_e/m_e)/clight
my_constants.n_macroparticles = 500
# Peak density for the transverse Gaussian particle initialization.
my_constants.plasma_density_m3 = 1.0e17
my_constants.axial_thickness = 0.1
my_constants.effective_cross_section_area = 2.0*pi*sigma_r*sigma_r
my_constants.n_real_particles = plasma_density_m3*effective_cross_section_area*axial_thickness

my_constants.f = 1e9
my_constants.Vrf = 1.0e3
my_constants.Omega = 2.0*pi*f
my_constants.omega_pe = sqrt(plasma_density_m3*q_e**2/(epsilon0*m_e))
my_constants.electron_plasma_period = 2.0*pi/omega_pe
my_constants.max_dt = electron_plasma_period/10.0
my_constants.re = r0/10.0
my_constants.re2 = re*re

### Constants for 16 circular electrodes ###
my_constants.e0x = 0.0011000000000000001
my_constants.e0z = 0
my_constants.e1x = 0.0010162674857624155
my_constants.e1z = 0.00042095177560159876
my_constants.e2x = 0.00077781745930520236
my_constants.e2z = 0.00077781745930520225
my_constants.e3x = 0.00042095177560159887
my_constants.e3z = 0.0010162674857624155
my_constants.e4x = 0
my_constants.e4z = 0.0011000000000000001
my_constants.e5x = -0.00042095177560159871
my_constants.e5z = 0.0010162674857624155
my_constants.e6x = -0.00077781745930520225
my_constants.e6z = 0.00077781745930520236
my_constants.e7x = -0.0010162674857624155
my_constants.e7z = 0.00042095177560159893

####### BOUNDARY CONDITION ######
#################################

#################################
########## ELECTRODES ###########
#################################
# The positive-inside implicit function is the union of 16 electrode disks.
# Project y is WarpX z, so the electrode centers are stored as e*i*x/e*i*z.
warpx.eb_implicit_function = "max(max(max(max(re2 - ((x - e0x)**2 + (z - e0z)**2), re2 - ((x - e1x)**2 + (z - e1z)**2)), max(re2 - ((x - e2x)**2 + (z - e2z)**2), re2 - ((x - e3x)**2 + (z - e3z)**2))), max(max(re2 - ((x - e4x)**2 + (z - e4z)**2), re2 - ((x - e5x)**2 + (z - e5z)**2)), max(re2 - ((x - e6x)**2 + (z - e6z)**2), re2 - ((x - e7x)**2 + (z - e7z)**2)))), max(max(max(re2 - ((x - e8x)**2 + (z - e8z)**2), re2 - ((x - e9x)**2 + (z - e9z)**2)), max(re2 - ((x - e10x)**2 + (z - e10z)**2), re2 - ((x - e11x)**2 + (z - e11z)**2))), max(max(re2 - ((x - e12x)**2 + (z - e12z)**2), re2 - ((x - e13x)**2 + (z - e13z)**2)), max(re2 - ((x - e14x)**2 + (z - e14z)**2), re2 - ((x - e15x)**2 + (z - e15z)**2)))))"

# Alternating RF conducting electrodes: even i are +Vrf*cos(Omega*t), odd i are
# -Vrf*cos(Omega*t). The value outside electrodes is not used as an EB boundary.
warpx.eb_potential(x,y,z,t) = "if(((x - e0x)**2 + (z - e0z)**2) < re2, Vrf*cos(Omega*t), if(((x - e1x)**2 + (z - e1z)**2) < re2, -Vrf*cos(Omega*t), if(((x - e2x)**2 + (z - e2z)**2) < re2, Vrf*cos(Omega*t), if(((x - e3x)**2 + (z - e3z)**2) < re2, -Vrf*cos(Omega*t), if(((x - e4x)**2 + (z - e4z)**2) < re2, Vrf*cos(Omega*t), if(((x - e5x)**2 + (z - e5z)**2) < re2, -Vrf*cos(Omega*t), if(((x - e6x)**2 + (z - e6z)**2) < re2, Vrf*cos(Omega*t), if(((x - e7x)**2 + (z - e7z)**2) < re2, -Vrf*cos(Omega*t), if(((x - e8x)**2 + (z - e8z)**2) < re2, Vrf*cos(Omega*t), if(((x - e9x)**2 + (z - e9z)**2) < re2, -Vrf*cos(Omega*t), if(((x - e10x)**2 + (z - e10z)**2) < re2, Vrf*cos(Omega*t), if(((x - e11x)**2 + (z - e11z)**2) < re2, -Vrf*cos(Omega*t), if(((x - e12x)**2 + (z - e12z)**2) < re2, Vrf*cos(Omega*t), if(((x - e13x)**2 + (z - e13z)**2) < re2, -Vrf*cos(Omega*t), if(((x - e14x)**2 + (z - e14z)**2) < re2, Vrf*cos(Omega*t), if(((x - e15x)**2 + (z - e15z)**2) < re2, -Vrf*cos(Omega*t), 0.0))))))))))))))))"
